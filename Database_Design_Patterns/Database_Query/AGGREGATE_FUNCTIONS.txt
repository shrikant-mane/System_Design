-- COUNT()
SELECT
	DEPARTMENT,
	COUNT(*) AS COUNT
FROM EMPLOYEE
GROUP BY DEPARTMENT;


-- SUM()
SELECT
	DEPARTMENT,
	SUM(SALARY) "SUM SALARY"
FROM EMPLOYEE
GROUP BY DEPARTMENT;


-- MAX()
SELECT
	DEPARTMENT,
	MAX(SALARY) "MAX SALARY"
FROM EMPLOYEE
GROUP BY DEPARTMENT;


-- MIN()
SELECT
	DEPARTMENT,
	MIN(SALARY) "MIN SALARY"
FROM EMPLOYEE
GROUP BY DEPARTMENT;


-- AVG()
SELECT
	DEPARTMENT,
	ROUND(AVG(SALARY), 2) "AVG SALARY"
FROM EMPLOYEE
GROUP BY DEPARTMENT;



--COUNT()
-- COUNT(* | [DISTINCT] ALL | column_name)
SELECT COUNT(*) FROM STUDENT;

SELECT COUNT(CITY) FROM STUDENT;

SELECT COUNT(DISTINCT CITY) FROM STUDENT;

SELECT COUNT(DISTINCT CITY) AS "NUMBER OF CITIES" FROM STUDENT;

SELECT 
	COUNT(*) AS STUDENT,
	CITY
FROM STUDENT
WHERE MARKS > 70
GROUP BY CITY
HAVING COUNT(*) > 10
ORDER BY COUNT(*) DESC;



-- SUM()
-- SUM (* | [DISTINCT] ALL | column_name)

SELECT SUM(MARKS) FROM STUDENT;

SELECT SUM(DISTINCT MARKS) AS "TOTAL MARKS" FROM STUDENT;

SELECT 
	SUM(MARKS) AS MARKS,
	CITY
FROM STUDENT
WHERE MARKS > 70
GROUP BY CITY
HAVING SUM(MARKS) > 500
ORDER BY SUM(MARKS) DESC;



-- MAX()
-- MAX (* | [DISTINCT] ALL | column_name)
SELECT MAX(MARKS) FROM STUDENT;

SELECT MAX( DISTINCT MARKS) FROM STUDENT;

SELECT 
	MAX(MARKS) AS "MAX MARKS",
	CITY
FROM STUDENT 
GROUP BY CITY
ORDER BY MAX(MARKS) DESC
LIMIT 1;



-- MIN()
-- MIN (* | [DISTINCT] ALL | column_name)
SELECT MIN(MARKS) FROM STUDENT;

SELECT MIN( DISTINCT MARKS) FROM STUDENT;

SELECT 
	MIN(MARKS) AS "MIN MARKS",
	CITY
FROM STUDENT 
GROUP BY CITY
ORDER BY MIN(MARKS) DESC
LIMIT 1;



-- AVG()
-- AVG (* | [DISTINCT] ALL | column_name)

SELECT ROUND(AVG(MARKS), 2) FROM STUDENT;

SELECT ROUND(AVG(DISTINCT MARKS), 2) FROM STUDENT;

SELECT 
	ROUND(AVG(MARKS), 2) AS "AVG MARKS",
	CITY
FROM STUDENT 
GROUP BY CITY
ORDER BY AVG(MARKS) DESC
LIMIT 1;
